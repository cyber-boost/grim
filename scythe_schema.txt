-- Scythe License System Database Schema
-- White-Label Licensing Platform with RSA Signing & Hardware Fingerprinting
-- Database: scythe_licensing.db

-- Enable foreign keys for referential integrity
PRAGMA foreign_keys = ON;

-- ============================================================================
-- VENDORS TABLE - White-label customers
-- ============================================================================
CREATE TABLE IF NOT EXISTS vendors (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    vendor_id TEXT NOT NULL UNIQUE,
    company_name TEXT NOT NULL,
    contact_email TEXT NOT NULL,
    contact_name TEXT,
    api_key TEXT NOT NULL UNIQUE,
    api_secret TEXT NOT NULL,
    stripe_account_id TEXT,
    webhook_url TEXT,
    status TEXT DEFAULT 'active',
    plan_type TEXT DEFAULT 'basic',
    monthly_quota INTEGER DEFAULT 1000,
    current_usage INTEGER DEFAULT 0,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- ============================================================================
-- PRODUCTS TABLE - Software products managed by vendors
-- ============================================================================
CREATE TABLE IF NOT EXISTS products (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    product_id TEXT NOT NULL UNIQUE,
    vendor_id TEXT NOT NULL,
    product_name TEXT NOT NULL,
    product_version TEXT DEFAULT '1.0.0',
    description TEXT,
    license_template TEXT,
    features TEXT, -- JSON array
    pricing TEXT, -- JSON object
    status TEXT DEFAULT 'active',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (vendor_id) REFERENCES vendors(vendor_id)
);

-- ============================================================================
-- LICENSE KEYS TABLE - Generated licenses with RSA signing
-- ============================================================================
CREATE TABLE IF NOT EXISTS license_keys (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    license_key TEXT NOT NULL UNIQUE,
    product_id TEXT NOT NULL,
    vendor_id TEXT NOT NULL,
    customer_email TEXT NOT NULL,
    customer_name TEXT,
    hardware_fingerprint TEXT,
    features TEXT, -- JSON array
    expires_at TIMESTAMP,
    max_activations INTEGER DEFAULT 1,
    current_activations INTEGER DEFAULT 0,
    status TEXT DEFAULT 'active',
    payment_status TEXT DEFAULT 'pending',
    stripe_payment_id TEXT,
    rsa_signature TEXT NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    activated_at TIMESTAMP,
    last_validated TIMESTAMP,
    FOREIGN KEY (product_id) REFERENCES products(product_id),
    FOREIGN KEY (vendor_id) REFERENCES vendors(vendor_id)
);

-- ============================================================================
-- ACTIVATIONS TABLE - Track license activations
-- ============================================================================
CREATE TABLE IF NOT EXISTS activations (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    license_key TEXT NOT NULL,
    hardware_fingerprint TEXT NOT NULL,
    ip_address TEXT,
    user_agent TEXT,
    os_info TEXT,
    app_version TEXT,
    activation_token TEXT,
    status TEXT DEFAULT 'active',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    last_seen TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (license_key) REFERENCES license_keys(license_key)
);

-- ============================================================================
-- VIOLATIONS TABLE - License violation tracking
-- ============================================================================
CREATE TABLE IF NOT EXISTS violations (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    license_key TEXT NOT NULL,
    violation_type TEXT NOT NULL,
    hardware_fingerprint TEXT,
    ip_address TEXT,
    details TEXT,
    severity TEXT DEFAULT 'medium',
    user_agent TEXT,
    os_info TEXT,
    reported_by TEXT DEFAULT 'system',
    action_taken TEXT,
    resolved BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (license_key) REFERENCES license_keys(license_key)
);

-- ============================================================================
-- API USAGE TABLE - Track API calls for billing
-- ============================================================================
CREATE TABLE IF NOT EXISTS api_usage (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    vendor_id TEXT NOT NULL,
    endpoint TEXT NOT NULL,
    method TEXT NOT NULL,
    response_code INTEGER,
    request_size INTEGER DEFAULT 0,
    response_size INTEGER DEFAULT 0,
    processing_time_ms INTEGER DEFAULT 0,
    ip_address TEXT,
    user_agent TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (vendor_id) REFERENCES vendors(vendor_id)
);

-- ============================================================================
-- PAYMENTS TABLE - Stripe payment tracking
-- ============================================================================
CREATE TABLE IF NOT EXISTS payments (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    payment_id TEXT NOT NULL UNIQUE,
    vendor_id TEXT NOT NULL,
    license_key TEXT,
    stripe_payment_intent_id TEXT,
    stripe_charge_id TEXT,
    amount_cents INTEGER NOT NULL,
    currency TEXT DEFAULT 'usd',
    status TEXT DEFAULT 'pending',
    customer_email TEXT,
    metadata TEXT, -- JSON
    webhook_received_at TIMESTAMP,
    processed_at TIMESTAMP,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (vendor_id) REFERENCES vendors(vendor_id),
    FOREIGN KEY (license_key) REFERENCES license_keys(license_key)
);

-- ============================================================================
-- WEBHOOKS TABLE - Webhook delivery tracking
-- ============================================================================
CREATE TABLE IF NOT EXISTS webhooks (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    vendor_id TEXT NOT NULL,
    event_type TEXT NOT NULL,
    webhook_url TEXT NOT NULL,
    payload TEXT NOT NULL,
    response_code INTEGER,
    response_body TEXT,
    retry_count INTEGER DEFAULT 0,
    max_retries INTEGER DEFAULT 3,
    status TEXT DEFAULT 'pending',
    scheduled_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    delivered_at TIMESTAMP,
    FOREIGN KEY (vendor_id) REFERENCES vendors(vendor_id)
);

-- ============================================================================
-- PERFORMANCE INDEXES
-- ============================================================================

-- Vendors table indexes
CREATE INDEX IF NOT EXISTS idx_vendors_api_key ON vendors(api_key);
CREATE INDEX IF NOT EXISTS idx_vendors_status ON vendors(status);

-- Products table indexes
CREATE INDEX IF NOT EXISTS idx_products_vendor_id ON products(vendor_id);
CREATE INDEX IF NOT EXISTS idx_products_status ON products(status);

-- License keys table indexes
CREATE INDEX IF NOT EXISTS idx_license_keys_product_id ON license_keys(product_id);
CREATE INDEX IF NOT EXISTS idx_license_keys_vendor_id ON license_keys(vendor_id);
CREATE INDEX IF NOT EXISTS idx_license_keys_customer_email ON license_keys(customer_email);
CREATE INDEX IF NOT EXISTS idx_license_keys_status ON license_keys(status);
CREATE INDEX IF NOT EXISTS idx_license_keys_expires_at ON license_keys(expires_at);

-- Activations table indexes
CREATE INDEX IF NOT EXISTS idx_activations_license_key ON activations(license_key);
CREATE INDEX IF NOT EXISTS idx_activations_hardware_fingerprint ON activations(hardware_fingerprint);
CREATE INDEX IF NOT EXISTS idx_activations_status ON activations(status);

-- Violations table indexes
CREATE INDEX IF NOT EXISTS idx_violations_license_key ON violations(license_key);
CREATE INDEX IF NOT EXISTS idx_violations_type ON violations(violation_type);
CREATE INDEX IF NOT EXISTS idx_violations_severity ON violations(severity);
CREATE INDEX IF NOT EXISTS idx_violations_created_at ON violations(created_at);

-- API usage table indexes
CREATE INDEX IF NOT EXISTS idx_api_usage_vendor_id ON api_usage(vendor_id);
CREATE INDEX IF NOT EXISTS idx_api_usage_created_at ON api_usage(created_at);
CREATE INDEX IF NOT EXISTS idx_api_usage_endpoint ON api_usage(endpoint);

-- Payments table indexes
CREATE INDEX IF NOT EXISTS idx_payments_vendor_id ON payments(vendor_id);
CREATE INDEX IF NOT EXISTS idx_payments_status ON payments(status);
CREATE INDEX IF NOT EXISTS idx_payments_stripe_payment_intent_id ON payments(stripe_payment_intent_id);

-- Webhooks table indexes
CREATE INDEX IF NOT EXISTS idx_webhooks_vendor_id ON webhooks(vendor_id);
CREATE INDEX IF NOT EXISTS idx_webhooks_status ON webhooks(status);
CREATE INDEX IF NOT EXISTS idx_webhooks_scheduled_at ON webhooks(scheduled_at);
